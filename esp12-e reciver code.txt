// =================================================
// Convert String CSV packet into values
// Packet format:
// ID,TEMP,HUMIDITY,GAS,VIBRATION,FLAME
// =================================================

bool parsePacket(String data) {

  data.trim();

  Serial.println();
  Serial.print("Received packet: ");
  Serial.println(data);

  int p1 = data.indexOf(',');

  if (p1 == -1)
    return false;

  int p2 = data.indexOf(',', p1 + 1);

  if (p2 == -1)
    return false;

  int p3 = data.indexOf(',', p2 + 1);

  if (p3 == -1)
    return false;

  int p4 = data.indexOf(',', p3 + 1);

  if (p4 == -1)
    return false;

  int p5 = data.indexOf(',', p4 + 1);

  if (p5 == -1)
    return false;

  // Extract values

  packetID = data.substring(0, p1).toInt();

  temperature = data.substring(p1 + 1, p2).toFloat();

  humidity = data.substring(p2 + 1, p3).toFloat();

  gasValue = data.substring(p3 + 1, p4).toInt();

  vibration = data.substring(p4 + 1, p5).toInt();

  flame = data.substring(p5 + 1).toInt();

  return true;
}

// =================================================
// SETUP
// =================================================

void setup() {

  Serial.begin(115200);

  delay(1000);

  Serial.println();
  Serial.println("===============================");
  Serial.println(" LoRa Industrial Monitoring");
  Serial.println(" ESP-12E RECEIVER");
  Serial.println("===============================");

  pinMode(BUZZER_PIN, OUTPUT);

  digitalWrite(BUZZER_PIN, LOW);

  // ================= WiFi =================

  WiFi.mode(WIFI_STA);

  WiFi.begin(ssid, password);

  Serial.print("Connecting to WiFi");

  while (WiFi.status() != WL_CONNECTED) {

    delay(500);

    Serial.print(".");
  }

  Serial.println();

  Serial.println("WiFi connected!");

  Serial.print("IP Address: ");
  Serial.println(WiFi.localIP());

  // ================= LoRa =================

  SPI.begin();

  LoRa.setPins(
    LORA_SS,
    LORA_RST,
    LORA_DIO0
  );

  Serial.println("Starting LoRa...");

  if (!LoRa.begin(LORA_FREQUENCY)) {

    Serial.println("ERROR: LoRa initialization failed!");

    while (1) {
      delay(1000);
    }
  }

  LoRa.enableCrc();

  LoRa.setSpreadingFactor(7);

  LoRa.setSignalBandwidth(125E3);

  LoRa.setCodingRate4(5);

  Serial.println("LoRa initialized successfully.");

  // ================= Web server =================

  server.on("/", HTTP_GET,
    [](AsyncWebServerRequest *request) {

      request->send(
        200,
        "text/html",
        htmlPage()
      );

    }
  );

  server.begin();

  Serial.println("Web server started.");

  Serial.println();
  Serial.println("Open the IP address shown above.");
}

// =================================================
// LOOP
// =================================================

void loop() {

  int packetSize = LoRa.parsePacket();

  if (packetSize) {

    String receivedData = "";

    while (LoRa.available()) {

      receivedData +=
        (char)LoRa.read();
    }

    rssiValue = LoRa.packetRssi();

    Serial.println();
    Serial.println("===============================");

    if (parsePacket(receivedData)) {

      packetReceived = true;

      Serial.println("DATA PARSED SUCCESSFULLY");

      Serial.print("Packet ID: ");
      Serial.println(packetID);

      Serial.print("Temperature: ");
      Serial.println(temperature);

      Serial.print("Humidity: ");
      Serial.println(humidity);

      Serial.print("Gas: ");
      Serial.println(gasValue);

      Serial.print("Vibration: ");
      Serial.println(vibration);

      Serial.print("Flame: ");
      Serial.println(flame);

      Serial.print("RSSI: ");
      Serial.println(rssiValue);

      // ================= ALARM =================

      bool alarm = false;

      if (flame == 1)
        alarm = true;

      if (vibration == 1)
        alarm = true;

      if (gasValue > 2000)
        alarm = true;

      if (temperature > 40)
        alarm = true;

      if (alarm) {

        digitalWrite(BUZZER_PIN, HIGH);

        Serial.println("!!! SYSTEM ALARM !!!");

      } else {

        digitalWrite(BUZZER_PIN, LOW);

        Serial.println("SYSTEM NORMAL");
      }

    } else {

      Serial.println("ERROR: Invalid packet format");
      Serial.println("Expected:");
      Serial.println("ID,TEMP,HUMIDITY,GAS,VIBRATION,FLAME");
    }

    Serial.println("===============================");
  }
}
